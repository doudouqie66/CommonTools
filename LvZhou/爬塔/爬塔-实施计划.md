-- 环形层管理：螺旋上升路径、安全区平台、层数追踪

local RingManager = {
    CurrentRing = 0,
    TotalRings = 20,
    PlayersInSafeZone = {},     -- { playerKey = true/false }
    PlayerCurrentRing = {},     -- { playerKey = ringNum }
}

function RingManager:Init(totalRings)
    self.TotalRings = totalRings or self.TotalRings
    self.CurrentRing = 1
    ugcprint("RingManager:Init totalRings=" .. tostring(self.TotalRings))
end

-- 进入下一圈
function RingManager:EnterNextRing(playerKey)
    local nextRing = (self.PlayerCurrentRing[playerKey] or 0) + 1
    if nextRing > self.TotalRings then
        self:OnReachedTop(playerKey)
        return
    end
    self.PlayerCurrentRing[playerKey] = nextRing
    self.CurrentRing = nextRing
    UGCEventSystem:SendEvent(TowerEvents.RingEntered, playerKey, nextRing)
    ugcprint("RingManager:EnterNextRing playerKey=" .. tostring(playerKey) .. " ring=" .. tostring(nextRing))
end

-- 玩家进入安全区
function RingManager:EnterSafeZone(playerKey)
    self.PlayersInSafeZone[playerKey] = true
    ugcprint("RingManager:EnterSafeZone playerKey=" .. tostring(playerKey))
end

-- 玩家离开安全区
function RingManager:LeaveSafeZone(playerKey)
    self.PlayersInSafeZone[playerKey] = false
    ugcprint("RingManager:LeaveSafeZone playerKey=" .. tostring(playerKey))
end

-- 是否在安全区内
function RingManager:IsInSafeZone(playerKey)
    return self.PlayersInSafeZone[playerKey] == true
end

-- 到达塔顶
function RingManager:OnReachedTop(playerKey)
    UGCEventSystem:SendEvent(TowerEvents.ReachedTop, playerKey)
    ugcprint("RingManager:OnReachedTop playerKey=" .. tostring(playerKey))
end

-- 重置（被怪物抓到）
function RingManager:ResetPlayer(playerKey)
    self.PlayerCurrentRing[playerKey] = 0
    self.PlayersInSafeZone[playerKey] = false
    ugcprint("RingManager:ResetPlayer playerKey=" .. tostring(playerKey))
end

-- 获取玩家当前层数
function RingManager:GetPlayerRing(playerKey)
    return self.PlayerCurrentRing[playerKey] or 0

return RingManager

**Files:**

```lua
-- Script/Tower/EventManager.lua
-- 定时事件循环：按固定时间轴触发全局事件

UGCGameSystem.UGCRequire('Script.Tower.TowerDefines')

local EventManager = {
    EventSchedule = {           -- 时间轴（秒）
        { time=60,  eventType=GlobalEventType.SpeedDown,   warningDuration=10,  eventDuration=10 },
        { time=120, eventType=GlobalEventType.Night,       warningDuration=10,  eventDuration=10 },
        { time=300, eventType=GlobalEventType.SpeedUp,     warningDuration=10,  eventDuration=15 },
        { time=450, eventType=GlobalEventType.ReverseCtrl, warningDuration=10,  eventDuration=10 },
        { time=600, eventType=GlobalEventType.Night,       warningDuration=5,   eventDuration=5 },
    },
    CycleDuration = 600,        -- 循环周期（秒）
    ElapsedTime = 0,
    TimerHandle = nil,
    CurrentEventIndex = 1,
    CurrentPhase = nil,         -- "warning" / "active" / nil
}

function EventManager:Init()
    ugcprint("EventManager:Init")
end

-- 开始事件循环
function EventManager:StartLoop()
    self.ElapsedTime = 0
    self.CurrentEventIndex = 1
    self:ScheduleNextEvent()
end

-- 安排下一个事件
function EventManager:ScheduleNextEvent()
    local event = self.EventSchedule[self.CurrentEventIndex]
    if not event then
        -- 循环：时间到了回到第一个事件
        self.CurrentEventIndex = 1
        event = self.EventSchedule[1]
    end

    -- 计算距离事件触发的时间
    local waitTime = event.time - (self.ElapsedTime % self.CycleDuration)
    if waitTime <= 0 then waitTime = waitTime + self.CycleDuration end

    ugcprint("EventManager:ScheduleNextEvent event=" .. tostring(event.eventType) .. " in " .. tostring(waitTime) .. "s")

    -- 倒计时阶段
    self.TimerHandle = UGCTimerUtility.CreateLuaTimer(waitTime - event.warningDuration, function()
        self:EnterWarningPhase(event)
    end, false)
end

-- 进入倒计时阶段
function EventManager:EnterWarningPhase(event)
    self.CurrentPhase = "warning"
    UGCEventSystem:SendEvent(TowerEvents.EventWarning, event.eventType, event.warningDuration)
    ugcprint("EventManager:Warning " .. tostring(event.eventType) .. " for " .. tostring(event.warningDuration) .. "s")

    -- 倒计时结束 → 进入生效阶段
    self.TimerHandle = UGCTimerUtility.CreateLuaTimer(event.warningDuration, function()
        self:EnterActivePhase(event)
    end, false)
end

-- 进入生效阶段
function EventManager:EnterActivePhase(event)
    self.CurrentPhase = "active"
    UGCEventSystem:SendEvent(TowerEvents.EventActive, event.eventType, event.eventDuration)
    ugcprint("EventManager:Active " .. tostring(event.eventType) .. " for " .. tostring(event.eventDuration) .. "s")

    -- 生效结束
    self.TimerHandle = UGCTimerUtility.CreateLuaTimer(event.eventDuration, function()
        self:EndEvent(event)
    end, false)
end

-- 事件结束
function EventManager:EndEvent(event)
    self.CurrentPhase = nil
    UGCEventSystem:SendEvent(TowerEvents.EventEnded, event.eventType)
    ugcprint("EventManager:Ended " .. tostring(event.eventType))

    -- 安排下一个事件
    self.CurrentEventIndex = self.CurrentEventIndex + 1
    self:ScheduleNextEvent()
end

-- 暂停事件循环（玩家被捉时）
function EventManager:Pause()
    if self.TimerHandle then
        UGCTimerUtility.StopTimer(self.TimerHandle)
        self.TimerHandle = nil
    end
end

-- 恢复
function EventManager:Resume()
    self:ScheduleNextEvent()
end

function EventManager:Cleanup()
    self:Pause()
end

return EventManager
```

-- 给玩家添加道具
function ItemManager:AddItem(playerKey, itemId, count)
    local cfg = self.ItemConfig[itemId]
    if not cfg then
        ugcprint("ItemManager:AddItem unknown itemId=" .. tostring(itemId))
        return false
    end

    if not self.PlayerItems[playerKey] then
        self.PlayerItems[playerKey] = {}
    end

    local stack = self.PlayerItems[playerKey][itemId] or { count=0, durability=cfg.maxDurability }
    stack.count = (stack.count or 0) + (count or 1)

    -- 不超过最大持有数
    if cfg.maxCount > 0 and stack.count > cfg.maxCount then
        stack.count = cfg.maxCount
    end

    self.PlayerItems[playerKey][itemId] = stack
    ugcprint("ItemManager:AddItem playerKey=" .. tostring(playerKey) .. " item=" .. tostring(itemId) .. " count=" .. tostring(stack.count))
    return true
end

-- 使用道具
function ItemManager:UseItem(playerKey, itemId)
    local cfg = self.ItemConfig[itemId]
    if not cfg then return false end

    -- 安全检查区内禁用
    if not cfg.canUseInSafeZone then
        local inSafe = RingManager:IsInSafeZone(playerKey)
        if inSafe then
            ugcprint("ItemManager:UseItem blocked - in safe zone, item=" .. tostring(itemId))
            return false
        end
    end

    local stack = self.PlayerItems[playerKey] and self.PlayerItems[playerKey][itemId]
    if not stack or stack.count <= 0 then
        ugcprint("ItemManager:UseItem no stock, item=" .. tostring(itemId))
        return false
    end

    -- 扣除使用次数
    stack.count = stack.count - 1

    -- 处理耐久度（工具类）
    if cfg.category == ItemCategory.Tool and cfg.maxDurability > 0 then
        stack.durability = stack.durability - 1
        if stack.durability <= 0 then
            -- 耐久耗尽，道具损坏
            stack.count = 0
            UGCEventSystem:SendEvent(TowerEvents.ItemBroken, playerKey, itemId)
            ugcprint("ItemManager:Item broken playerKey=" .. tostring(playerKey) .. " item=" .. tostring(itemId))
        end
    end

    -- 发送使用事件
    UGCEventSystem:SendEvent(TowerEvents.ItemUsed, playerKey, itemId)
    ugcprint("ItemManager:UseItem playerKey=" .. tostring(playerKey) .. " item=" .. tostring(itemId))

    -- 清除空堆
